\olsection{पूर्ण संगणनीयतः परिगणनीय समुच्चय}

\begin{defn}
कोई समुच्चय $A$ (अनेक-एक अपचयनीयता के अंतर्गत) \emph{पूर्ण
!!{computably enumerable} समुच्चय} है, यदि
\begin{enumerate}
\item $A$ संगणनीयतः परिगणनीय है, और
\item किसी भी अन्य संगणनीयतः परिगणनीय समुच्चय $B$ के लिए $B \leq_m A$।
\end{enumerate}
\end{defn}

दूसरे शब्दों में, पूर्ण संगणनीयतः परिगणनीय समुच्चय संभवतः ``सबसे कठिन''
संगणनीयतः परिगणनीय समुच्चय हैं। वे \emph{किसी भी} संगणनीयतः परिगणनीय
समुच्चय के प्रश्नों का उत्तर देने देते हैं।

\begin{thm}
$K$, $K_0$, और $K_1$ सभी पूर्ण संगणनीयतः परिगणनीय समुच्चय हैं।
\end{thm}

\begin{proof}
$K_0$ को पूर्ण देखने के लिए $B$ को कोई संगणनीयतः परिगणनीय समुच्चय
मानें। तब किसी सूचकांक $e$ के लिए
\[
B = W_e = \Setabs{x}{\cfind{e}(x) \fdefined}.
\]
$f$ को फलन $f(x) = \tuple{e, x}$ मानें। तब प्रत्येक प्राकृतिक संख्या
$x$ के लिए $x \in B$ तभी और केवल तभी जब $f(x) \in K_0$। दूसरे शब्दों
में, $f$, $B$ को~$K_0$ में अपचयित करता है।

$K_1$ को पूर्ण देखने के लिए ध्यान दें कि \olref[k1]{prop:k1} के प्रमाण
में हमने $K_0$ को उसमें अपचयित किया था। अतः \olref[ppr]{prop:trans-red}
से किसी भी संगणनीयतः परिगणनीय समुच्चय को~$K_1$ में भी अपचयित किया जा
सकता है।

$K$ को लगभग इसी ढंग से $K_0$ में अपचयित किया जा सकता है।
\end{proof}

\begin{prob}
$K$ से $K_0$ में कोई अपचयन दें।
\end{prob}

\begin{digress}
इस प्रकार अब तक विचार किए गए संगणनीयतः परिगणनीय समुच्चयों के सभी उदाहरण
या तो संगणनीय हैं या पूर्ण। यह विचित्र लगना चाहिए!{} क्या ऐसे संगणनीयतः
परिगणनीय समुच्चय भी हैं जो न संगणनीय हों, न पूर्ण? उत्तर हाँ है, किंतु
1950 के दशक के मध्य में ही फ्रीडबर्ग और मुचनिक ने स्वतंत्र रूप से इसे
स्थापित किया।
\end{digress}
