\olsection{\usetoken{P}{formula}ों के समुच्चय से \usetoken{S}{derivability}}

\olref[prf][prs]{sec} में हमने किसी तंत्र~$\Sigma$ में !!{formula} की
प्रमाण्यता की धारणा परिभाषित की थी। अब हम इसे समुच्चय~$\Gamma$ के !!{formula}ों
से $\Sigma$ में प्रमाण्यता तक विस्तृत करते हैं।

\begin{defn}\ollabel{defn:Gammaproves}
  कोई !!{formula}~$!A$, तंत्र~$\Sigma$ में !!{formula}ों के समुच्चय $\Gamma$ से
  !!{derivable} है, जिसे $\Gamma \Proves[\Sigma] !A$ लिखते हैं, तभी जब
  ऐसे $!B_1$, \dots, $!B_n \in \Gamma$ हों कि
  $\Sigma \Proves !B_1 \lif (!B_2 \lif \cdots (!B_n \lif !A)
  \cdots)$. 
\end{defn}
