\olsection{अतिरिक्त नियमों के लिए यथार्थता}

हम किसी नियम को प्रतिरूपों के किसी वर्ग के लिए यथार्थ कहते हैं यदि, जब भी
किसी !!a{tableau} की कोई शाखा उस वर्ग के किसी प्रतिरूप में संतोषणीय हो, तब
नियम लगाने से मिली शाखा भी उस वर्ग के किसी प्रतिरूप में संतोषणीय हो।

\begin{prop}
  \ollabel{prop:soundness-T} 
  \iftag{prvBox}{\Ax{T}$\Box$}{}%
  \iftag{notprvBox,notprvDiamond}{}{ और }%
  \iftag{prvDiamond}{\Ax{T}$\Diamond$}{}
  स्वपरावर्ती प्रतिरूपों के लिए यथार्थ
  \iftag{notprvBox,notprvDiamond}{है}{हैं}।
\end{prop}

\begin{proof}
\begin{tagenumerate}{prvBox,prvDiamond}
\tagitem{prvBox}{शाखा को $\sFmla{\True}{\Box !B}[\sigma] \in \Gamma$
  पर T$\Box$ लगाकर विस्तृत किया जाता है:
  \iftag{probBox}{अभ्यास।}{इससे शाखा पर नया !!{signed formula}~$\sFmla{\True}{!B}[\sigma]$ मिलता है। मान लें
    $\mSat{M}{\Gamma}[f]$, विशेषतः
    $\mSat{M}{\Box !B}[f(\sigma)]$। चूँकि $R$ स्वपरावर्ती है, हम जानते हैं
    कि $Rf(\sigma)f(\sigma)$। अतः $\mSat{M}{!B}[f(\sigma)]$, अर्थात्
    $\mModel{M}, f$, $\sFmla{\True}{!B}[\sigma]$ को संतुष्ट करते हैं।}}{}
\tagitem{prvDiamond}{शाखा को $\sFmla{\False}{\Diamond !B}[\sigma] \in
  \Gamma$ पर T$\Diamond$ लगाकर विस्तृत किया जाता है:
  \iftag{probDiamond}{अभ्यास।}{इससे शाखा पर नया !!{signed formula}~$\sFmla{\False}{!B}[\sigma]$ मिलता है। मान लें
    $\mSat{M}{\Gamma}[f]$, विशेषतः
    $\mSat/{M}{\Diamond !B}[f(\sigma)]$। चूँकि $R$ स्वपरावर्ती है, हम
    जानते हैं कि $Rf(\sigma)f(\sigma)$। अतः
    $\mSat/{M}{!B}[f(\sigma)]$, अर्थात् $\mModel{M}, f$,
    $\sFmla{\False}{!B}[\sigma]$ को संतुष्ट करते हैं।}}{}
\end{tagenumerate}
\end{proof}

\tagprob{probBox,probDiamond}
\begin{prob}
  \olref[nml][tab][msn]{prop:soundness-T} का प्रमाण पूरा कीजिए।
\end{prob}
\tagendprob
