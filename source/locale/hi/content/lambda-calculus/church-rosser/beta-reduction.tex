\olsection{$\beta$-अपचयन}

\begin{lem}\ollabel{lem:one-par}
  यदि $M \bredone M'$, तो $M \bredpar M'$।
\end{lem}
\begin{proof} यदि $M \bredone M'$, तो किन्हीं $x$, $N$ और~$Q$ के लिए $M$,
    ~$(\lambd[x][N])Q$ और $M'$, ~$\Subst{N}{Q}{x}$ है। चूँकि
    \olref[pb]{thm:refl} से $N \bredpar N$ और $Q \bredpar Q$, इसलिए
    \olref[pb]{defn:bredpar}\olref[pb]{defn:bredpar4} से तुरंत
    $(\lambd[x][N])Q \bredpar \Subst{N}{Q}{x}$।
\end{proof}

\begin{lem}\ollabel{lem:par-red}
  यदि $M \bredpar M'$, तो $M \bred M'$।
\end{lem}

\begin{proof} $M \bredpar M'$ के !!{derivation} पर आगमन द्वारा।
  \begin{enumerate}
  \item अंतिम नियम \olref[pb]{defn:bredpar1} है: तब $M$ और $M'$ केवल
    $x$ हैं, और $x \bred x$।
  \item अंतिम नियम \olref[pb]{defn:bredpar2} है: किन्हीं $x$, $N$, $N'$
    के लिए $M$, ~$\lambd[x][N]$ और $M'$, ~$\lambd[x][N']$ है, जहाँ $N
    \bredpar N'$। आगमन-परिकल्पना से $N \bred N'$। तब $N \bred N'$ वाले
    $\bredone$ संकुचनों के उसी अनुक्रम से $\lambd[x][N] \bred
    \lambd[x][N']$।
  \item अंतिम नियम \olref[pb]{defn:bredpar3} है: किन्हीं $P$, $Q$, $P'$,
    $Q'$ के लिए $M$, ~$PQ$ और $M'$, ~$P'Q'$ है, जहाँ $P \bredpar P'$ और
    $Q \bredpar Q'$। आगमन-परिकल्पना से $P \bred P'$ और $Q \bred Q'$।
    अतः पहले $P \bred P'$ और फिर $Q \bred Q'$ वाले अपचयन अनुक्रम से $PQ
    \bred P'Q'$।
  \item अंतिम नियम \olref[pb]{defn:bredpar4} है: किन्हीं $x$, $N$, $M'$,
    $Q$, $Q'$ के लिए $M$, ~$(\lambd[x][N])Q$ और $M'$,
    ~$\Subst{N'}{Q'}{x}$ है, जहाँ $N \bredpar N'$ और $Q \bredpar Q'$।
    आगमन-परिकल्पना से $Q \bred Q'$ और $N \bred N'$। अतः पहले $N \bred
    N'$, फिर $Q \bred Q'$, और अंत में $(\lambd[x][N'])Q'$ को
    $\Subst{N'}{Q'}{x}$ में संकुचित करके $(\lambd[x][N])Q \bred
    \Subst{N'}{Q'}{x}$।
  \end{enumerate}
\end{proof}

\begin{lem}\ollabel{lem:str}
  $\bred$ वह सबसे छोटा संक्रामक संबंध है जिसमें $\bredpar$ समाहित है।
\end{lem}

\begin{proof}
  $\xred$ वह सबसे छोटा संक्रामक संबंध हो जिसमें $\bredpar$ समाहित है।

  $\bred \subseteq \xred$: मान लें $M \bred M'$, अर्थात् $M \ident M_1
  \bredone \dots \bredone M_k \ident M'$। \olref{lem:one-par} से $M
  \ident M_1 \bredpar \dots \bredpar M_k \ident M'$। चूँकि $\xred$ में
  $\bredpar$ समाहित है और वह संक्रामक है, इसलिए $M \xred M'$।

  $\xred \subseteq \bred$: मान लें $M \xred M'$, अर्थात् $M \ident M_1
  \bredpar \dots \bredpar M_k \ident M'$। \olref{lem:par-red} से $M
  \ident M_1 \bred \dots \bred M_k \ident M'$। चूँकि $\bred$ संक्रामक
  है, इसलिए $M \bred M'$।
\end{proof}

\begin{thm}\ollabel{thm:cr}
  $\bred$ चर्च--रॉसर गुण संतुष्ट करता है।
\end{thm}

\begin{proof}
  \olref[dap]{thm:str}, \olref[pb]{thm:cr} और \olref{lem:str} से तात्कालिक।
\end{proof}
